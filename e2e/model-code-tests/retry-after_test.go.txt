package candidate

import (
	"net/http"
	"testing"
	"time"
)

func TestRetryAfter(t *testing.T) {
	now := time.Date(2026, 10, 22, 12, 0, 0, 0, time.UTC)
	for _, c := range []struct {
		s    string
		want time.Duration
		ok   bool
	}{
		{"0", 0, true}, {"120", 120 * time.Second, true}, {"  007  ", 7 * time.Second, true},
		{now.Add(91 * time.Second).Format(http.TimeFormat), 91 * time.Second, true},
		{now.Add(-time.Second).Format(http.TimeFormat), 0, true},
		{"", 0, false}, {"-1", 0, false}, {"+2", 0, false}, {"1.5", 0, false}, {"1e2", 0, false}, {"invalid", 0, false},
		{"9223372036", 9223372036 * time.Second, true}, {"18446744074", 0, false}, {"9223372037", 0, false}, {"18446744073709551616", 0, false},
	} {
		got, err := ParseRetryAfter(c.s, now)
		if (err == nil) != c.ok || c.ok && got != c.want {
			t.Errorf("ParseRetryAfter(%q)=(%v,%v), want (%v,ok=%v)", c.s, got, err, c.want, c.ok)
		}
	}
}
