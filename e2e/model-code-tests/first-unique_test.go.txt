package candidate

import "testing"

func TestFirstUnique(t *testing.T) {
	cases := []struct {
		in   []string
		want string
		ok   bool
	}{
		{nil, "", false}, {[]string{}, "", false}, {[]string{"a", "a"}, "", false},
		{[]string{"a", "b", "a", "c", "c"}, "b", true},
		{[]string{"", "a", "a"}, "", true}, {[]string{"é", "文", "é", "z"}, "文", true},
		{[]string{"first", "second"}, "first", true},
	}
	for _, c := range cases {
		copyIn := append([]string(nil), c.in...)
		got, ok := FirstUnique(c.in)
		if got != c.want || ok != c.ok {
			t.Errorf("FirstUnique(%q)=(%q,%v), want (%q,%v)", c.in, got, ok, c.want, c.ok)
		}
		for i := range c.in {
			if c.in[i] != copyIn[i] {
				t.Fatal("input mutated")
			}
		}
	}
}
